# Домашнее задание к работе 8
## Условие задачи
Вычислить с использованием цикла for значение ряда для заданных пользователем значений a и n:

\[a(a+1)\cdots(a+n+1)\]
## 1.Алгоритм
1.**Начало**
2.Получение a, n
3.Вычисления
4.Вывод результата
## 2. Реализация программы
#include <stdio.h>
#include <locale.h>
int main()
{
	setlocale(LC_CTYPE, "RUS");
	int a, n;
	printf("Введите последовательно числа через запятую\n");
	scanf("%d,%d", &a, &n);
	int i;
	int res = 1;
	for (i = a+1; i <= n+1; i=i+1) 
	{
		res*= a * i;
		printf("%d\n", res);
	}
	printf("Итоговое значение вычислений = %d", res);
}
